CREATE DATABASE task_management;
USE task_management;
CREATE TABLE employee (
    sr_no INT AUTO_INCREMENT PRIMARY KEY,
    emp_id VARCHAR(20) NOT NULL UNIQUE,
    emp_name VARCHAR(100) NOT NULL,
    email VARCHAR(100) NOT NULL UNIQUE,
    phone VARCHAR(15) UNIQUE,
    emp_password VARCHAR(255) NOT NULL,
    department ENUM(
        'HEAD',
        'IT',
        'SALES'
    ) NOT NULL,
    role ENUM(
        'SUPER_ADMIN',
        'ADMIN',
        'PROJECT_MANAGER',
        'TEAM_LEADER',
        'DEVELOPER',
        'TESTER'
    ) NOT NULL,
    status ENUM(
        'ACTIVE',
        'INACTIVE',
        'SUSPENDED'
    ) NOT NULL DEFAULT 'ACTIVE',
    manager_id VARCHAR(20) NULL,
    otp VARCHAR(10) DEFAULT NULL,
    expire_at DATETIME DEFAULT NULL,
    deleted_at DATETIME DEFAULT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (manager_id) REFERENCES employee(emp_id)
);

CREATE TABLE project (
    id INT AUTO_INCREMENT PRIMARY KEY,
    project_id VARCHAR(20) unique NOT NULL,

    name VARCHAR(200) NOT NULL,
    description TEXT,

    created_by VARCHAR(20) NOT NULL,
    pm_id VARCHAR(20) NOT NULL,

    status ENUM('PLANNING','ACTIVE','COMPLETED') DEFAULT 'PLANNING',
    deadline DATETIME NULL,

    FOREIGN KEY (created_by) REFERENCES employee(emp_id),
    FOREIGN KEY (pm_id) REFERENCES employee(emp_id)
);

CREATE TABLE team (
    team_id varchar(25) PRIMARY KEY,
    project_id VARCHAR(20) NOT NULL,
    team_leader_id varchar(20) NOT NULL,

    FOREIGN KEY (project_id) REFERENCES project(project_id),
    FOREIGN KEY (team_leader_id) REFERENCES employee(emp_id),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE team_members (
    team_id varchar(30),
    employee_id varchar(30),

    PRIMARY KEY (team_id, employee_id),

    FOREIGN KEY (team_id) REFERENCES team(team_id),
    FOREIGN KEY (employee_id) REFERENCES employee(emp_id)
);

CREATE TABLE tasks (
    id INT AUTO_INCREMENT PRIMARY KEY,

    project_id varchar(30) NOT NULL,
    team_id varchar(30) NOT NULL,

    title VARCHAR(255) NOT NULL,
    description TEXT,

    assigned_to varchar(30) NOT NULL,
    created_by varchar(30) NOT NULL,

    status ENUM(
        'TODO',
        'IN_PROGRESS',
        'DEV_DONE',
        'TESTING',
        'COMPLETED'
    ) DEFAULT 'TODO',

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,
        deadline DATETIME NULL,


    FOREIGN KEY (project_id) REFERENCES project(project_id),
    FOREIGN KEY (team_id) REFERENCES team(team_id),
    FOREIGN KEY (assigned_to) REFERENCES employee(emp_id),
    FOREIGN KEY (created_by) REFERENCES employee(emp_id)
);
